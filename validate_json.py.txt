
import json
import re
from datetime import datetime
from google.colab import files

# GSTIN Validation
def validate_gstin(gstin):
    if not isinstance(gstin, str):
        return False

    pattern = r'^[0-9]{2}[A-Z]{5}[0-9]{4}[A-Z][1-9A-Z]Z[0-9A-Z]$'

    return bool(re.fullmatch(pattern, gstin))


# Invoice Number Validation
def validate_invoice_number(invoice_number):
    return (
        isinstance(invoice_number, str)
        and bool(invoice_number.strip())
    )


# Invoice Date Validation
def validate_invoice_date(invoice_date):
    if not isinstance(invoice_date, str):
        return False

    try:
        datetime.strptime(invoice_date, "%Y-%m-%d")
        return True
    except ValueError:
        return False


# Amount Validation
def validate_amount(amount):
    if isinstance(amount, bool):
        return False

    try:
        return float(amount) > 0 and float(amount) != float("inf")
    except (ValueError, TypeError, OverflowError):
        return False


# Main JSON Validation
def validate_json_file(file_path):

    try:
        with open(file_path, "r", encoding="utf-8") as file:
            data = json.load(file)

    except json.JSONDecodeError as e:
        print("Invalid JSON file:", e)
        return

    except Exception as e:
        print("File reading error:", e)
        return

    if not isinstance(data, dict):
        print("Invalid format: JSON must contain an invoice object.")
        return

    errors = []

    # Check required fields
    required_fields = [
        "gstin",
        "invoice_number",
        "invoice_date",
        "amount"
    ]

    for field in required_fields:
        if field not in data:
            errors.append(f"Missing required field: {field}")

    # Validate GSTIN
    if "gstin" in data:
        if not validate_gstin(data["gstin"]):
            errors.append("Invalid GSTIN. Must follow the 15-character GSTIN format.")

    # Validate Invoice Number
    if "invoice_number" in data:
        if not validate_invoice_number(data["invoice_number"]):
            errors.append("Invalid or empty invoice number.")

    # Validate Invoice Date
    if "invoice_date" in data:
        if not validate_invoice_date(data["invoice_date"]):
            errors.append("Invalid invoice date. Use YYYY-MM-DD format.")

    # Validate Amount
    if "amount" in data:
        if not validate_amount(data["amount"]):
            errors.append("Invalid amount. Enter a positive number.")

    # Display Results
    print("\n========== JSON VALIDATION REPORT ==========")

    if errors:
        print("\nVALIDATION FAILED!")

        for error in errors:
            print("❌", error)

    else:
        print("\nVALIDATION SUCCESSFUL!")
        print("✅ GSTIN:", data["gstin"])
        print("✅ Invoice Number:", data["invoice_number"])
        print("✅ Invoice Date:", data["invoice_date"])
        print("✅ Amount:", data["amount"])

    print("\n============================================")


# Upload JSON File in Google Colab
print("Please upload your JSON file:")

uploaded = files.upload()

# Read and Validate Uploaded File
for file_name in uploaded.keys():
    validate_json_file(file_name)
